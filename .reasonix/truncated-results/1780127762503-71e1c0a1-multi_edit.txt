+ ### 10.3 P1：SessionBase 模板基类消除重复
+ 
+ **目标**：`Session`（359 行）和 `TlsSession`（244 行）共享 ~300 行协议逻辑。提取为模板基类 `SessionBase<Stream>`。
+ 
+ ```cpp
+ // net/SessionBase.hpp（新文件，约 200 行共享逻辑）
+ #pragma once
+ #include <array>
+ #include <cstdint>
+ #include <deque>
+ #include <string>
+ #include <vector>
+ #include <boost/asio.hpp>
+ #include "net/Session.hpp"  // ISession, ISessionHandler
+ #include "net/Framing.hpp"
+ #include "util/DebugLog.hpp"
+ 
+ namespace net {
+ 
+ inline constexpr std::size_t kMaxJsonLineBytes = 1024 * 1024;
+ inline constexpr std::size_t kMaxBinaryPayloadBytes = 16 * 1024 * 1024;
+ 
+ template <typename Stream>
+ class SessionBase : public ISession {
+ public:
+     SessionBase(Stream stream, ISessionHandler& handler)
+         : stream_(std::move(stream)),
+           strand_(boost::asio::make_strand(stream_.lowest_layer().get_executor())),
+           handler_(handler) {}
+ 
+     void write_line(std::string text) override {
+         auto self = this->shared_from_this();
+         boost::asio::post(strand_, [self, t = std::move(text)]() mutable {
+             if (self->closed_) return;
+             while (!t.empty() && (t.back() == '\n' || t.back() == '\r')) t.pop_back();
+             t.push_back('\n');
+             self->enqueue_write(std::vector<char>(t.begin(), t.end()));
+         });
+     }
+ 
+     void write_binary(std::vector<char> payload) override {
+         auto self = this->shared_from_this();
+         boost::asio::post(strand_, [self, p = std::move(payload)]() mutable {
+             if (self->closed_ || p.size() > kMaxBinaryPayloadBytes) return;
+             auto len32 = static_cast<std::uint32_t>(p.size());
+             std::uint8_t be[4];
+             encode_be32(len32, be);
+             std::vector<char> packet(be, be + 4);
+             packet.insert(packet.end(), p.begin(), p.end());
+             self->enqueue_write(std::move(packet));
+         });
+     }
+ 
+     void begin_expect_binary(std::uint32_t byte_count) override {
+         auto self = this->shared_from_this();
+         boost::asio::post(strand_, [self, byte_count]() {
+             if (self->closed_ || self->rx_mode_ != RxMode::Lines) return;
+             if (byte_count > kMaxBinaryPayloadBytes) { self->fail("binary_expect_too_large"); return; }
+             self->rx_mode_ = RxMode::Binary;
+             self->binary_remaining_ = byte_count;
+             self->binary_accum_.clear();
+             self->binary_accum_.reserve(byte_count);
+             if (byte_count == 0) self->complete_binary_message();
+             else self->consume_recv_into_binary();
+         });
+     }
+ 
+     void request_close(const char* reason) override {
+         auto self = this->shared_from_this();
+         boost::asio::post(strand_, [self, r = reason ? reason : "request_close"]() {
+             if (!self->closed_) self->fail(r);
+         });
+     }
+ 
+     strand_type& strand() override { return strand_; }
+ 
+ protected:
+     void process_received_bytes(const char* data, std::size_t len) {
+         if (!append_bytes(recv_buffer_, data, len)) { fail("recv_buffer_overflow"); return; }
+         if (rx_mode_ == RxMode::Binary) consume_recv_into_binary();
+         else process_line_frames();
+     }
+ 
+     void process_line_frames() {
+         while (!closed_ && rx_mode_ == RxMode::Lines) {
+             PopLineResult r = try_pop_line(recv_buffer_);
+             if (r.status == PopLineResult::Status::None) break;
+             if (r.status == PopLineResult::Status::TooLong) { fail("line_too_long"); return; }
+             if (!r.line.empty()) { handler_.on_line(*this, std::move(r.line)); if (closed_) return; }
+         }
+     }
+ 
+     void consume_recv_into_binary() {
+         while (!closed_ && binary_remaining_ > 0 && !recv_buffer_.empty()) {
+             std::size_t take = std::min(recv_buffer_.size(), static_cast<std::size_t>(binary_remaining_));
+             binary_accum_.insert(binary_accum_.end(), recv_buffer_.begin(), recv_buffer_.begin() + take);
+             recv_buffer_.erase(0, take);
+             binary_remaining_ -= static_cast<std::uint32_t>(take);
+         }
+         if (!closed_ && binary_remaining_ == 0) complete_binary_message();
+     }
+ 
+     void complete_binary_message() {
+         rx_mode_ = RxMode::Lines; binary_remaining_ = 0;
+         std::vector<char> payload = std::move(binary_accum_); binary_accum_.clear();
+         handler_.on_binary(*this, std::move(payload));
+         if (!closed_ && !recv_buffer_.empty()) process_line_frames();
+     }
+ 
+     void enqueue_write(std::vector<char> chunk) {
+         if (closed_ || chunk.empty()) return;
+         bool idle = !write_in_progress_;
+         write_queue_.push_back(std::move(chunk));
+         if (idle) { write_in_progress_ = true; do_write_front(); }
+     }
+ 
+     void fail(const char* reason) {
+         if (closed_) return;
+         closed_ = true; reset_writes(); reset_rx_for_close();
+         boost::system::error_code ec; stream_.lowest_layer().close(ec);
+         handler_.on_error(*this, reason); handler_.on_close(*this);
+     }
+ 
+     void clean_close() {
+         if (closed_) return;
+         closed_ = true; reset_writes(); reset_rx_for_close();
+         boost::system::error_code ec; stream_.lowest_layer().close(ec);
+         handler_.on_close(*this);
+     }
+ 
+     virtual void do_read() = 0;
+     virtual void do_write_front() = 0;
+ 
+     Stream stream_;
+     boost::asio::strand<typename Stream::lowest_layer_type::executor_type> strand_;
+     ISessionHandler& handler_;
+     std::string recv_buffer_;
+     std::array<char, 8192> read_buf_{};
+     std::deque<std::vector<char>> write_queue_;
+     bool write_in_progress_ = false;
+     enum class RxMode { Lines, Binary };
+     RxMode rx_mode_ = RxMode::Lines;
+     std::uint32_t binary_remaining_ = 0;
+     std::vector<char> binary_accum_;
+     bool closed_ = false;
+ };
+ 
+ } // namespace net
+ ```
+ 
+ ```cpp
+ // net/Session.hpp — 重构后（~55 行，仅保留 TCP 读/写适配）
+ #pragma once
+ #include "net/SessionBase.hpp"
+ 
+ namespace net {
+ class Session : public SessionBase<boost::asio::ip::tcp::socket> {
+ public:
+     using SessionBase::SessionBase;
+     void start() override { if (!closed_) do_read(); }
+     boost::asio::ip::tcp::socket& socket() override { return stream_; }
+     const boost::asio::ip::tcp::socket& socket() const override { return stream_; }
+ protected:
+     void do_read() override {
+         auto self = std::static_pointer_cast<Session>(shared_from_this());
+         stream_.async_read_some(boost::asio::buffer(read_buf_),
+             boost::asio::bind_executor(strand_, [self](auto& ec, std::size_t n) {
+                 if (self->closed_) return;
+                 if (ec) { self->clean_close(); return; }
+                 self->process_received_bytes(self->read_buf_.data(), n);
+                 if (!self->closed_) self->do_read();
+             }));
+     }
+     void do_write_front() override {
+         if (closed_ || write_queue_.empty()) { write_in_progress_ = false; return; }
+         auto self = std::static_pointer_cast<Session>(shared_from_this());
+         auto& buf = write_queue_.front();
+         boost::asio::async_write(stream_, boost::asio::buffer(buf),
+             boost::asio::bind_executor(strand_, [self](auto& ec, std::size_t) {
+                 if (self->closed_) { self->reset_writes(); return; }
+                 if (ec) { self->fail("tcp_write_error"); return; }
+                 self->write_queue_.pop_front(); self->do_write_front();
+             }));
+     }
+ };
+ } // namespace net
+ ```

+ ### 10.6 P3：数据库连接池
+ 
+ **策略**：用预建连接池替换 `thread_local` 模式，连接数 = 配置的 `pool_size`，可小于线程数。
+ 
+ ```cpp
+ // db/ConnectionPool.hpp（新文件）
+ #pragma once
+ #include <condition_variable>
+ #include <memory>
+ #include <mutex>
+ #include <queue>
+ 
+ #include <mysqlx/xdevapi.h>
+ #include "db/DbConfig.hpp"
+ #include "db/DbClient.hpp"
+ 
+ namespace db {
+ 
+ class ConnectionPool {
+ public:
+     ConnectionPool(const DbConfig& cfg, std::size_t pool_size)
+         : cfg_(cfg), max_size_(std::max<std::size_t>(1, pool_size)) {
+         for (std::size_t i = 0; i < max_size_; ++i)
+             pool_.push(std::make_unique<mysqlx::Session>(DbClient::connectionUrl(cfg)));
+     }
+ 
+     std::unique_ptr<mysqlx::Session> acquire() {
+         std::unique_lock<std::mutex> lk(mu_);
+         cv_.wait(lk, [this] { return !pool_.empty(); });
+         auto sess = std::move(pool_.front());
+         pool_.pop();
+         return sess;
+     }
+ 
+     void release(std::unique_ptr<mysqlx::Session> sess) {
+         if (!sess) return;
+         try { sess->sql("SELECT 1").execute(); }  // 快速健康检查
+         catch (...) {
+             try { sess = std::make_unique<mysqlx::Session>(DbClient::connectionUrl(cfg_)); }
+             catch (...) { return; }
+         }
+         std::lock_guard<std::mutex> lk(mu_);
+         pool_.push(std::move(sess));
+         cv_.notify_one();
+     }
+ 
+ private:
+     DbConfig cfg_;
+     std::size_t max_size_;
+     mutable std::mutex mu_;
+     std::condition_variable cv_;
+     std::queue<std::unique_ptr<mysqlx::Session>> pool_;
+ };
+ 
+ } // namespace db
+ ```

+ ```cpp
+ // service/DbExecutor.cpp — 适配连接池（仅改 dispatch 方法）
+ void DbExecutor::dispatch(Work work, Done done) {
+     boost::asio::post(*pool_, [this, work = std::move(work),
+                                done = std::move(done)]() mutable {
+         auto session = connection_pool_->acquire();
+         std::exception_ptr ep;
+         try { work(*session); }
+         catch (...) { ep = std::current_exception(); }
+         connection_pool_->release(std::move(session));
+         if (static_cast<bool>(done)) {
+             try { done(ep); }
+             catch (...) { /* 同上 */ }
+         }
+     });
+ }
+ ```
+ 
+ ### 10.7 P3：错误处理统一（AppError）
+ 
+ **策略**：不引入 `Result<T>`（避免双轨制），用 `AppError` 继承 `std::runtime_error` + 错误码，现有 try-catch 全部兼容。
+ 
+ ```cpp
+ // protocol/AppError.hpp（新文件）
+ #pragma once
+ #include <stdexcept>
+ #include <string>
+ 
+ namespace protocol {
+ 
+ enum class ErrorCode {
+     Success = 0,
+     ParseError = 1001,  PayloadTooLarge = 1002,
+     NotAuthenticated = 2001,  AuthFailed = 2002,
+     TokenInvalid = 2003,  TokenExpired = 2004,
+     Forbidden = 3001,
+     EmployeeNotFound = 4001,  DeviceNotFound = 4002,
+     DbError = 5001,  ArcFaceError = 6001,
+ };
+ 
+ class AppError : public std::runtime_error {
+ public:
+     AppError(ErrorCode code, std::string message)
+         : std::runtime_error(message), code_(code) {}
+     ErrorCode code() const { return code_; }
+     int http_style_code() const { return static_cast<int>(code_); }
+ private:
+     ErrorCode code_;
+ };
+ 
+ } // namespace protocol
+ ```
+ 
+ ```cpp
+ // 使用示例 — Repository 层
+ void PersonRepository::insert(mysqlx::Session& session, const PersonCreateInput& input) {
+     if (input.name.empty() || input.employee_id.empty())
+         throw protocol::AppError(ErrorCode::ParseError, "name and employee_id are required");
+     try {
+         // ... MySQL 操作 ...
+     } catch (const mysqlx::Error& e) {
+         throw protocol::AppError(ErrorCode::DbError,
+             std::string("Person insert failed: ") + e.what());
+     }
+ }
+ ```
